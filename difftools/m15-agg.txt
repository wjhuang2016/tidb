# Matrix 15: aggregates — bit aggs, GROUP_CONCAT semantics, decimal scale, statistical
>>>
CREATE TABLE ag(id INT, g INT, v INT, d DECIMAL(10,4), s VARCHAR(10), b BLOB);
INSERT INTO ag VALUES (1,1,3,1.5,'aa','x'),(2,1,7,2.25,'bb','y'),(3,2,NULL,3.125,'AA','z'),(4,2,5,NULL,'cc','w'),(5,3,0,0.0001,'ab','q');
SELECT BIT_AND(v), BIT_OR(v), BIT_XOR(v) FROM ag;
SELECT BIT_AND(v), BIT_OR(v), BIT_XOR(v) FROM ag WHERE g = 2;
SELECT BIT_AND(NULL) FROM ag;
SELECT SUM(v), AVG(v), COUNT(v), COUNT(*) FROM ag;
SELECT AVG(d), SUM(d) FROM ag;
SELECT STD(v), STDDEV(v), STDDEV_POP(v), VARIANCE(v), VAR_POP(v), STDDEV_SAMP(v), VAR_SAMP(v) FROM ag;
SELECT GROUP_CONCAT(s) FROM ag;
SELECT GROUP_CONCAT(s ORDER BY id DESC) FROM ag;
SELECT GROUP_CONCAT(s ORDER BY s SEPARATOR '; ') FROM ag;
SELECT GROUP_CONCAT(DISTINCT UPPER(s)) FROM ag;
SELECT GROUP_CONCAT(v ORDER BY v) FROM ag;
SELECT GROUP_CONCAT(DISTINCT g ORDER BY g DESC) FROM ag;
SELECT GROUP_CONCAT(s, v) FROM ag;
SELECT MAX(d), MIN(s), MAX(s) FROM ag;
SELECT APPROX_COUNT_DISTINCT(g) FROM ag;
SELECT APPROX_PERCENTILE(v, 50) FROM ag;
SELECT g, COUNT(*), GROUP_CONCAT(id) FROM ag GROUP BY g HAVING COUNT(*) > 0 ORDER BY g;
SELECT SUM(ALL v), SUM(DISTINCT v) FROM ag;
SELECT COUNT(DISTINCT g, s) FROM ag;
SELECT JSON_ARRAYAGG(v) FROM ag;
SELECT JSON_OBJECTAGG(g, v) FROM ag;
SELECT ANY_VALUE(s) FROM ag WHERE g = 3;
SELECT GROUPING(g), g FROM ag GROUP BY g WITH ROLLUP;
SELECT g, GROUP_CONCAT(s ORDER BY g) FROM ag GROUP BY g ORDER BY g DESC;
