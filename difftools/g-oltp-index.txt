-- OLTP battery 1: index selection and plan divergence
-- Focus: composite index, covering index, index merge, order/limit

DROP DATABASE IF EXISTS oltp1;
CREATE DATABASE oltp1;
USE oltp1;

-- table with multiple indexes for selection testing
CREATE TABLE t1 (
  id INT PRIMARY KEY AUTO_INCREMENT,
  a INT,
  b INT,
  c VARCHAR(50),
  d DATETIME,
  KEY idx_a (a),
  KEY idx_b (b),
  KEY idx_ab (a, b),
  KEY idx_abc (a, b, c),
  UNIQUE KEY uk_c (c)
);

INSERT INTO t1 (a, b, c, d) VALUES
(1, 10, 'aaa', '2024-01-01 10:00:00'),
(1, 20, 'bbb', '2024-01-02 11:00:00'),
(2, 10, 'ccc', '2024-01-03 12:00:00'),
(2, 20, 'ddd', '2024-01-04 13:00:00'),
(3, 30, 'eee', '2024-01-05 14:00:00'),
(NULL, 40, 'fff', '2024-01-06 15:00:00'),
(4, NULL, 'ggg', '2024-01-07 16:00:00');

-- point lookup on composite index prefix
SELECT * FROM t1 WHERE a = 1;
SELECT * FROM t1 WHERE a = 1 AND b = 10;
SELECT * FROM t1 WHERE a = 1 AND b = 20 AND c = 'bbb';

-- range on composite index
SELECT * FROM t1 WHERE a = 2 AND b > 10;
SELECT * FROM t1 WHERE a > 1 AND b = 10;

-- covering index (idx_abc covers a,b,c - no table lookup needed)
SELECT a, b, c FROM t1 WHERE a = 1;
SELECT a, b FROM t1 WHERE a = 1 AND b > 5;

-- index merge: OR across different indexes
SELECT * FROM t1 WHERE a = 1 OR b = 30;
SELECT * FROM t1 WHERE a = 2 OR c = 'fff';

-- descending order with limit (common OLTP pagination)
SELECT * FROM t1 ORDER BY id DESC LIMIT 3;
SELECT * FROM t1 WHERE a > 0 ORDER BY id DESC LIMIT 2;

-- index hints
SELECT * FROM t1 USE INDEX (idx_a) WHERE a = 1;
SELECT * FROM t1 FORCE INDEX (idx_ab) WHERE a = 1 AND b = 10;
SELECT * FROM t1 IGNORE INDEX (idx_a) WHERE a = 1;

-- NULL handling in indexed columns
SELECT * FROM t1 WHERE a IS NULL;
SELECT * FROM t1 WHERE b IS NULL;
SELECT * FROM t1 WHERE a IS NULL OR b IS NULL;

-- unique index lookup
SELECT * FROM t1 WHERE c = 'aaa';
SELECT * FROM t1 WHERE c IN ('aaa', 'bbb', 'zzz');

-- EXPLAIN to check plan divergence
EXPLAIN SELECT * FROM t1 WHERE a = 1 OR b = 30;
EXPLAIN SELECT * FROM t1 WHERE a > 1 AND b = 10;
EXPLAIN SELECT a, b FROM t1 WHERE a = 1 AND b > 5;
