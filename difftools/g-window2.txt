# window frames expanded
CREATE TABLE w2(g INT, v DECIMAL(10,2), s VARCHAR(5), d DATE);
INSERT INTO w2 VALUES (1,1.5,'a','2020-01-01'),(1,2.5,'b','2020-01-02'),(2,3.5,'c','2020-01-03'),(2,3.5,'d','2020-01-04'),(3,NULL,'e','2020-01-05');
>>>
SELECT s, SUM(v) OVER (ORDER BY d ROWS BETWEEN UNBOUNDED PRECEDING AND UNBOUNDED FOLLOWING) FROM w2;
SELECT s, SUM(v) OVER (ORDER BY d RANGE BETWEEN INTERVAL 1 DAY PRECEDING AND INTERVAL 1 DAY FOLLOWING) FROM w2;
SELECT s, SUM(v) OVER (ORDER BY d RANGE BETWEEN INTERVAL 1 DAY PRECEDING AND CURRENT ROW) FROM w2;
SELECT s, SUM(v) OVER (ORDER BY d ROWS BETWEEN 1 PRECEDING AND UNBOUNDED FOLLOWING) FROM w2;
SELECT s, COUNT(v) OVER (ORDER BY d), COUNT(*) OVER (ORDER BY d) FROM w2;
SELECT s, SUM(v) OVER (PARTITION BY g ORDER BY d ROWS BETWEEN 1 FOLLOWING AND 2 FOLLOWING) FROM w2;
SELECT s, LAG(s, 0) OVER (ORDER BY d) FROM w2;
SELECT s, LAG(v, -1) OVER (ORDER BY d) FROM w2;
SELECT s, LEAD(v, 10, 0) OVER (ORDER BY d) FROM w2;
SELECT s, NTH_VALUE(v, 1) OVER (ORDER BY d) FROM w2;
SELECT s, NTH_VALUE(v, 99) OVER (ORDER BY d) FROM w2;
SELECT s, NTH_VALUE(s, 2) FROM w2;
SELECT s, NTILE(0) OVER (ORDER BY d) FROM w2;
SELECT s, NTILE(-1) OVER (ORDER BY d) FROM w2;
SELECT s, NTILE(100) OVER (ORDER BY d) FROM w2;
SELECT s, PERCENT_RANK() OVER (PARTITION BY g ORDER BY d), CUME_DIST() OVER (PARTITION BY g ORDER BY d) FROM w2;
SELECT s, SUM(v) OVER w, AVG(v) OVER w2 FROM w2 WINDOW w AS (ORDER BY d), w2 AS (PARTITION BY g);
SELECT s, ROW_NUMBER() OVER (ORDER BY s), RANK() OVER (ORDER BY v), DENSE_RANK() OVER (ORDER BY v) FROM w2;
SELECT g, SUM(v) FROM w2 GROUP BY g WINDOW w AS (ORDER BY g);
SELECT DISTINCT g, SUM(v) OVER (PARTITION BY g) FROM w2;
SELECT s FROM w2 ORDER BY SUM(v) OVER ();
SELECT s, SUM(v) OVER (ORDER BY d RANGE BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) FROM w2;
SELECT s, SUM(v) OVER (ORDER BY v RANGE BETWEEN CURRENT ROW AND CURRENT ROW) FROM w2;
SELECT s, SUM(v) OVER (ORDER BY d ROWS BETWEEN CURRENT ROW AND 1 PRECEDING) FROM w2;
SELECT s, SUM(v) OVER (ORDER BY d ROWS BETWEEN 5 FOLLOWING AND 2 FOLLOWING) FROM w2;
SELECT s, (SELECT SUM(v) OVER () FROM w2 LIMIT 1) FROM w2 LIMIT 1;
