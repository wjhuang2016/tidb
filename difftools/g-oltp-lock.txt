-- OLTP battery 2: pessimistic locking and FOR UPDATE
-- Focus: SELECT FOR UPDATE, NOWAIT, SKIP LOCKED, lock conflicts

DROP DATABASE IF EXISTS oltp2;
CREATE DATABASE oltp2;
USE oltp2;

CREATE TABLE t1 (
  id INT PRIMARY KEY,
  val INT,
  name VARCHAR(50)
);

INSERT INTO t1 VALUES (1, 100, 'alice'), (2, 200, 'bob'), (3, 300, 'charlie');

-- basic FOR UPDATE
BEGIN;
SELECT * FROM t1 WHERE id = 1 FOR UPDATE;
COMMIT;

-- FOR UPDATE on range
BEGIN;
SELECT * FROM t1 WHERE id > 1 FOR UPDATE;
COMMIT;

-- FOR UPDATE NOWAIT (should succeed when no conflict)
BEGIN;
SELECT * FROM t1 WHERE id = 2 FOR UPDATE NOWAIT;
COMMIT;

-- FOR UPDATE SKIP LOCKED (should succeed when no conflict)
BEGIN;
SELECT * FROM t1 WHERE id IN (1, 2, 3) FOR UPDATE SKIP LOCKED;
COMMIT;

-- UPDATE with WHERE (pessimistic lock acquisition)
BEGIN;
UPDATE t1 SET val = val + 1 WHERE id = 1;
SELECT * FROM t1 WHERE id = 1;
COMMIT;

-- DELETE with WHERE
BEGIN;
DELETE FROM t1 WHERE id = 3;
SELECT COUNT(*) FROM t1;
ROLLBACK;

-- FOR UPDATE with ORDER BY LIMIT
BEGIN;
SELECT * FROM t1 ORDER BY id DESC LIMIT 1 FOR UPDATE;
COMMIT;

-- FOR UPDATE with IN clause
BEGIN;
SELECT * FROM t1 WHERE id IN (1, 2) FOR UPDATE;
COMMIT;

-- check lock behavior with non-existent rows
BEGIN;
SELECT * FROM t1 WHERE id = 999 FOR UPDATE;
COMMIT;

-- FOR UPDATE on secondary index lookup
CREATE TABLE t2 (
  id INT PRIMARY KEY,
  uk INT UNIQUE,
  val VARCHAR(20)
);
INSERT INTO t2 VALUES (1, 10, 'x'), (2, 20, 'y');

BEGIN;
SELECT * FROM t2 WHERE uk = 10 FOR UPDATE;
COMMIT;

BEGIN;
SELECT * FROM t2 WHERE uk IN (10, 20) FOR UPDATE NOWAIT;
COMMIT;

-- check REPEATABLE READ snapshot behavior
SET SESSION TRANSACTION ISOLATION LEVEL REPEATABLE READ;
BEGIN;
SELECT * FROM t1 WHERE id = 1;
COMMIT;

-- check READ COMMITTED behavior
SET SESSION TRANSACTION ISOLATION LEVEL READ COMMITTED;
BEGIN;
SELECT * FROM t1 WHERE id = 1;
COMMIT;

SET SESSION TRANSACTION ISOLATION LEVEL REPEATABLE READ;
