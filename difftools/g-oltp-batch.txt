-- OLTP battery 6: batch INSERT and transaction rollback
-- Focus: multi-value INSERT, ROLLBACK behavior, partial success

DROP DATABASE IF EXISTS oltp6;
CREATE DATABASE oltp6;
USE oltp6;

CREATE TABLE t1 (
  id INT PRIMARY KEY,
  val INT
);

-- batch insert success
BEGIN;
INSERT INTO t1 VALUES (1, 10), (2, 20), (3, 30);
SELECT * FROM t1 ORDER BY id;
COMMIT;

SELECT * FROM t1 ORDER BY id;

-- batch insert with rollback
BEGIN;
INSERT INTO t1 VALUES (4, 40), (5, 50);
SELECT * FROM t1 ORDER BY id;
ROLLBACK;

SELECT * FROM t1 ORDER BY id;

-- partial conflict in batch insert (should fail entire statement)
BEGIN;
INSERT INTO t1 VALUES (6, 60), (1, 70), (7, 80);
COMMIT;

SELECT * FROM t1 ORDER BY id;

-- batch insert with auto-commit off
SET autocommit = 0;
INSERT INTO t1 VALUES (8, 80);
SELECT * FROM t1 WHERE id = 8;
ROLLBACK;
SELECT * FROM t1 WHERE id = 8;
SET autocommit = 1;

-- large batch insert (100 rows)
BEGIN;
INSERT INTO t1 VALUES 
  (100, 1), (101, 2), (102, 3), (103, 4), (104, 5),
  (105, 6), (106, 7), (107, 8), (108, 9), (109, 10),
  (110, 11), (111, 12), (112, 13), (113, 14), (114, 15),
  (115, 16), (116, 17), (117, 18), (118, 19), (119, 20);
SELECT COUNT(*) FROM t1 WHERE id >= 100;
COMMIT;

-- UPDATE with transaction
BEGIN;
UPDATE t1 SET val = val + 1000 WHERE id < 10;
SELECT * FROM t1 WHERE id < 10 ORDER BY id;
ROLLBACK;

SELECT * FROM t1 WHERE id < 10 ORDER BY id;

-- DELETE with transaction
BEGIN;
DELETE FROM t1 WHERE id >= 100;
SELECT COUNT(*) FROM t1;
ROLLBACK;

SELECT COUNT(*) FROM t1;

-- mixed DML in transaction
BEGIN;
INSERT INTO t1 VALUES (200, 200);
UPDATE t1 SET val = 999 WHERE id = 1;
DELETE FROM t1 WHERE id = 2;
SELECT * FROM t1 WHERE id IN (1, 2, 200) ORDER BY id;
ROLLBACK;

SELECT * FROM t1 WHERE id IN (1, 2, 200) ORDER BY id;

-- savepoint (if supported)
BEGIN;
INSERT INTO t1 VALUES (300, 300);
SAVEPOINT sp1;
INSERT INTO t1 VALUES (301, 301);
SELECT COUNT(*) FROM t1 WHERE id >= 300;
ROLLBACK TO SAVEPOINT sp1;
SELECT COUNT(*) FROM t1 WHERE id >= 300;
COMMIT;

SELECT COUNT(*) FROM t1 WHERE id >= 300;
