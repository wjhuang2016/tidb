# Matrix 10: string builtin breadth
>>>
SELECT CHAR(65,66,67), CHAR(65,66,67 USING utf8mb4), CHAR(256, 65536);
SELECT HEX(CHAR(0x7F, 0x80, 0xFF));
SELECT BIN(11), OCT(11), CONV('ff', 16, 2), CONV('zz', 36, 10), CONV(-1, 10, 16);
SELECT FORMAT(1234567.891, 2), FORMAT(1234567.891, 0), FORMAT('abc', 1);
SELECT INSERT('hello', 2, 3, 'XY'), INSERT('hello', -1, 3, 'XY'), INSERT('hello', 2, -1, 'XY');
SELECT INSTR('hello', 'll'), LOCATE('l', 'hello', 4), POSITION('' IN 'abc');
SELECT ELT(2, 'a', 'b', 'c'), ELT(4, 'a', 'b'), FIELD('b', 'a', 'b', 'c'), FIELD('z', 'a', 'b');
SELECT FIND_IN_SET('b', 'a,b,c'), FIND_IN_SET('b', 'abc');
SELECT MAKE_SET(5, 'a', 'b', 'c');
SELECT EXPORT_SET(5, 'Y', 'N', ',', 4);
SELECT LOAD_FILE('/etc/hostname');
SELECT QUOTE('a\b''c'), QUOTE(NULL);
SELECT UNQUOTE('a');
SELECT LEFT('hello', 100), RIGHT('hello', 100), SUBSTRING('hello', -3), SUBSTRING('hello', 3, -1);
SELECT MID('hello', 2, 2);
SELECT REVERSE('abc'), REVERSE('中文');
SELECT SPACE(3), SPACE(-1), REPEAT('ab', 3);
SELECT RPAD('ab', 5, '123456'), RPAD('中文', 5, 'x');
SELECT TRIM('  x  '), TRIM(BOTH 'x' FROM 'xxx'), TRIM(LEADING 'x' FROM 'xxa'), TRIM(TRAILING 'x' FROM 'axx');
SELECT LTRIM('  x'), RTRIM('x  ');
SELECT UPPER('abcß'), LOWER('ABC'), LCASE('AbC'), UCASE('aBc');
SELECT REPLACE('aaa', 'aa', 'b'), REPLACE('abc', '', 'x');
SELECT STRCMP('a', 'b'), STRCMP('a', 'a'), STRCMP('2', '10');
SELECT SOUNDEX('Hello'), SOUNDEX('Quadratically'), SOUNDEX('');
SELECT 'abc' SOUNDS LIKE 'abc';
SELECT SUBSTRING_INDEX('a.b.c', '.', 2), SUBSTRING_INDEX('a.b.c', '.', -2), SUBSTRING_INDEX('abc', '.', 2);
SELECT TO_BASE64('abc'), FROM_BASE64('YWJj'), FROM_BASE64('!!!');
SELECT TO_BASE64('中文');
SELECT MD5(''), SHA1(''), SHA2('', 256), SHA2('', 224), SHA2('', 512);
SELECT AES_ENCRYPT('text', 'key');
SELECT COMPRESS('aaaaaaaaaa'), UNCOMPRESS(COMPRESS('aaaaaaaaaa'));
SELECT CRC32('abc'), CRC32('');
SELECT PASSWORD('x');
SELECT ENCODE('a','k');
SELECT WEIGHT_STRING('abc' AS BINARY(5));
SELECT BIT_LENGTH('中'), OCTET_LENGTH('中');
SELECT SUBSTR('abcdef', 2);
SELECT ASCII(''), ASCII(NULL);
SELECT CONCAT_WS('-', 'a', NULL, 'b'), CONCAT_WS(NULL, 'a');
SELECT CHARSET('abc'), COLLATION('abc');
