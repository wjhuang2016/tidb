# JSON functions deep-path battery
CREATE TABLE jt(a JSON, b INT);
INSERT INTO jt VALUES ('{"x":[1,2,{"y":"z"}],"k":null}', 1);
>>>
SELECT JSON_EXTRACT(a, '$.x[1]') FROM jt;
SELECT JSON_EXTRACT(a, '$.x[2].y') FROM jt;
SELECT JSON_EXTRACT(a, '$.x[*]') FROM jt;
SELECT JSON_EXTRACT(a, '$**.y') FROM jt;
SELECT JSON_EXTRACT(a, '$.x[1 to 2]') FROM jt;
SELECT JSON_EXTRACT(a, '$.x[last-1]') FROM jt;
SELECT JSON_EXTRACT(a, '$.nope') FROM jt;
SELECT JSON_EXTRACT(a, 1) FROM jt;
SELECT JSON_EXTRACT(a, '$.k'), JSON_TYPE(JSON_EXTRACT(a, '$.k')) FROM jt;
SELECT JSON_SET(a, '$.new', 5) FROM jt;
SELECT JSON_SET(a, '$.x[0]', JSON_OBJECT('q', 9)) FROM jt;
SELECT JSON_INSERT(a, '$.x[0]', 0) FROM jt;
SELECT JSON_REPLACE(a, '$.nope', 1) FROM jt;
SELECT JSON_REMOVE(a, '$.x[0]') FROM jt;
SELECT JSON_MERGE_PATCH(a, '{"m":[1]}') FROM jt;
SELECT JSON_MERGE_PRESERVE(a, '{"k":[2]}') FROM jt;
SELECT JSON_ARRAY(1, 'a', NULL, a) FROM jt;
SELECT JSON_OBJECT('k', a, 'n', NULL) FROM jt;
SELECT JSON_QUOTE('a"b'), JSON_UNQUOTE('"\\"a\\""');
SELECT JSON_CONTAINS(a, '1', '$.x') FROM jt;
SELECT JSON_CONTAINS_PATH(a, 'one', '$.x', '$.zz') FROM jt;
SELECT JSON_CONTAINS_PATH(a, 'all', '$.x', '$.zz') FROM jt;
SELECT JSON_KEYS(a), JSON_KEYS(a, '$.x[2]') FROM jt;
SELECT JSON_LENGTH(a), JSON_LENGTH(a, '$.x') FROM jt;
SELECT JSON_DEPTH(a), JSON_DEPTH('[]'), JSON_DEPTH('1') FROM jt;
SELECT JSON_TYPE(a), JSON_VALID(a), JSON_VALID('{bad') FROM jt;
SELECT JSON_SEARCH(a, 'all', 'z'), JSON_SEARCH(a, 'all', '%z%'), JSON_SEARCH(a, 'all', 'z', NULL, '$.x[2]') FROM jt;
SELECT JSON_PRETTY(a) FROM jt;
SELECT JSON_STORAGE_SIZE(a), JSON_STORAGE_FREE(a) FROM jt;
SELECT JSON_MEMBEROF(2, a->'$.x') FROM jt;
SELECT a->'$.x[1]', a->>'$.x[2].y' FROM jt;
SELECT a->'$."k"' FROM jt;
SELECT JSON_OVERLAPS('[1,2]','[2,3]'), JSON_OVERLAPS('[1]','"1"'), JSON_OVERLAPS('{"a":1}','{"a":1}');
SELECT JSON_SCHEMA_VALID('{"type":"object"}', a) FROM jt;
SELECT JSON_SCHEMA_VALID('{"type":"string"}', a) FROM jt;
SELECT JSON_TABLE(a, '$.x[*]' COLUMNS(v INT PATH '$')) AS t FROM jt;
UPDATE jt SET a = JSON_SET(a, '$.z', true);
SELECT a FROM jt;
SELECT JSON_EXTRACT('{"a":1}', '$.a') = 1;
SELECT CAST(a AS CHAR) FROM jt;
SELECT CAST(a AS DOUBLE), CAST(a AS UNSIGNED) FROM jt;
