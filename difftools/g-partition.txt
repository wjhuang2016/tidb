CREATE TABLE pa(a INT, b DATE, c VARCHAR(5));
>>>
ALTER TABLE pa PARTITION BY RANGE (a) (PARTITION p0 VALUES LESS THAN (10), PARTITION p1 VALUES LESS THAN (100));
INSERT INTO pa VALUES (5,'2020-01-01','x'),(50,'2020-01-01','y'),(500,'2020-01-01','z');
SELECT * FROM pa;
EXPLAIN SELECT * FROM pa WHERE a = 5;
EXPLAIN SELECT * FROM pa WHERE a > 5;
EXPLAIN FORMAT='brief' SELECT * FROM pa WHERE a = 50;
SELECT * FROM pa PARTITION (p0);
SELECT * FROM pa PARTITION (p1);
ALTER TABLE pa ADD PARTITION (PARTITION p2 VALUES LESS THAN (1000));
ALTER TABLE pa DROP PARTITION p2;
ALTER TABLE pa TRUNCATE PARTITION p1;
SELECT * FROM pa;
CREATE TABLE pb(a INT) PARTITION BY RANGE COLUMNS(a) (PARTITION p0 VALUES LESS THAN (5), PARTITION p1 VALUES LESS THAN (10));
CREATE TABLE pc(a INT, b VARCHAR(5)) PARTITION BY LIST COLUMNS(b) (PARTITION p0 VALUES IN ('x'), PARTITION p1 VALUES IN ('y'));
INSERT INTO pc VALUES (1,'x'),(2,'y');
SELECT * FROM pc PARTITION (p0);
INSERT INTO pc VALUES (3,'z');
CREATE TABLE pd(a INT, b INT) PARTITION BY LIST(a) (PARTITION p0 VALUES IN (1,2,3), PARTITION p1 VALUES IN (4,5));
INSERT INTO pd VALUES (1,'x'),(4,'y');
SELECT * FROM pd;
INSERT INTO pd VALUES (9,'z');
CREATE TABLE pe(a INT) PARTITION BY KEY(a) PARTITIONS 3;
INSERT INTO pe VALUES (1),(2),(3);
SELECT * FROM pe PARTITION (p0);
CREATE TABLE pf(a INT, b DATE) PARTITION BY RANGE COLUMNS(b) (PARTITION p0 VALUES LESS THAN ('2021-01-01'), PARTITION p1 VALUES LESS THAN ('2022-01-01'));
INSERT INTO pf VALUES (1,'2020-06-01'),(2,'2021-06-01');
EXPLAIN SELECT * FROM pf WHERE b = '2020-06-01';
ALTER TABLE pf REORGANIZE PARTITION p1 INTO (PARTITION p1 VALUES LESS THAN ('2021-06-01'), PARTITION p2 VALUES LESS THAN ('2022-01-01'));
SHOW CREATE TABLE pf;
CREATE TABLE pg(a INT) PARTITION BY HASH(a) PARTITIONS 2 SUBPARTITION BY HASH(a) SUBPARTITIONS 2;
CREATE TABLE ph(a INT) PARTITION BY RANGE (a) (PARTITION p0 VALUES LESS THAN (MAXVALUE));
ALTER TABLE ph REORGANIZE PARTITION p0 INTO (PARTITION p0 VALUES LESS THAN (10), PARTITION p1 VALUES LESS THAN MAXVALUE);
