CREATE TABLE p1(a INT, b VARCHAR(10)) PARTITION BY RANGE (a) (PARTITION p0 VALUES LESS THAN (10), PARTITION p1 VALUES LESS THAN (100));
INSERT INTO p1 VALUES (5, 'lo'), (50, 'hi'), (500, 'out');
SELECT * FROM p1 PARTITION (p0);
SELECT * FROM p1 PARTITION (p0, p1);
SELECT * FROM p1 WHERE a = 5;
EXPLAIN SELECT * FROM p1 WHERE a = 5;
EXPLAIN PARTITIONS SELECT * FROM p1 WHERE a < 20;
ALTER TABLE p1 ADD PARTITION (PARTITION p2 VALUES LESS THAN (1000));
ALTER TABLE p1 DROP PARTITION p2;
ALTER TABLE p1 TRUNCATE PARTITION p0;
SELECT * FROM p1;
ALTER TABLE p1 REORGANIZE PARTITION p0 INTO (PARTITION p0a VALUES LESS THAN (5), PARTITION p0b VALUES LESS THAN (10));
SHOW CREATE TABLE p1;
ALTER TABLE p1 COALESCE PARTITION 1;
CREATE TABLE p2(a INT) PARTITION BY HASH(a) PARTITIONS 4;
INSERT INTO p2 VALUES (1),(2),(3);
SELECT * FROM p2 PARTITION (p1);
EXPLAIN PARTITIONS SELECT * FROM p2 WHERE a = 3;
ALTER TABLE p2 COALESCE PARTITION 2;
CREATE VIEW v1 AS SELECT a, b FROM p1 WHERE a > 0;
SHOW CREATE VIEW v1;
SELECT * FROM v1;
INSERT INTO v1 VALUES (7, 'ins');
SELECT * FROM p1 WHERE a = 7;
CREATE OR REPLACE VIEW v1 AS SELECT a FROM p1;
SELECT * FROM v1;
CREATE VIEW v2 AS SELECT a FROM p1 UNION SELECT a FROM p1;
SELECT COUNT(*) FROM v2;
CREATE ALGORITHM = TEMPTABLE VIEW v3 AS SELECT DISTINCT a FROM p1;
SELECT * FROM v3;
CREATE VIEW v4(a_alias) AS SELECT a FROM p1;
SELECT a_alias FROM v4 WHERE a_alias = 5;
DROP VIEW v1, v2, v3, v4;
CREATE TEMPORARY TABLE tmp1(a INT);
INSERT INTO tmp1 VALUES (9);
SELECT * FROM tmp1;
SHOW TABLES LIKE 'tmp1';
CREATE TEMPORARY TABLE h1(a INT);
SELECT * FROM h1;
DROP TEMPORARY TABLE h1;
SELECT * FROM h1;
DROP TEMPORARY TABLE IF EXISTS tmp1;
CREATE TABLE gb1(a VARCHAR(10) CHARACTER SET gbk);

SELECT a, HEX(a), LENGTH(a), CHAR_LENGTH(a) FROM gb1;
SELECT CONVERT(a USING utf8mb4) FROM gb1;
