CREATE TABLE g1(a INT PRIMARY KEY, b INT);

INSERT INTO g1 VALUES (4, 40);

BEGIN;
UPDATE g1 SET b = 99 WHERE a = 1;
SAVEPOINT s1;
DELETE FROM g1 WHERE a = 2;

COMMIT;
SELECT * FROM g1;
