# Matrix 16: UNION type merging, window functions, subqueries
>>>
CREATE TABLE u1(a INT, b DECIMAL(10,2), c VARCHAR(5), d DATE);
INSERT INTO u1 VALUES (1, 1.5, 'x', '2020-01-01'), (2, 2.25, 'yy', '2021-06-15');
SELECT a FROM u1 UNION SELECT b FROM u1;
SELECT a FROM u1 UNION ALL SELECT c FROM u1;
SELECT b FROM u1 UNION SELECT d FROM u1;
SELECT c FROM u1 UNION SELECT d FROM u1;
SELECT a FROM u1 UNION SELECT NULL;
SELECT NULL UNION SELECT a FROM u1;
SELECT b FROM u1 UNION SELECT 3 FROM u1;
SELECT b FROM u1 UNION ALL SELECT 1.5e0 FROM u1;
SELECT CONCAT(c, '!') FROM u1 UNION SELECT a FROM u1;
SELECT a FROM u1 UNION SELECT b FROM u1 ORDER BY 1 DESC;
(SELECT a FROM u1 ORDER BY a LIMIT 1) UNION ALL (SELECT b FROM u1 ORDER BY b LIMIT 1);
SELECT COUNT(*) FROM (SELECT a FROM u1 UNION SELECT c FROM u1) t;
SELECT a, ROW_NUMBER() OVER (ORDER BY a) FROM u1;
SELECT a, RANK() OVER (ORDER BY b), DENSE_RANK() OVER (ORDER BY b) FROM u1;
SELECT a, SUM(a) OVER (ORDER BY a ROWS BETWEEN 1 PRECEDING AND CURRENT ROW) FROM u1;
SELECT a, LAG(a, 1, 0) OVER (ORDER BY a), LEAD(a) OVER (ORDER BY a) FROM u1;
SELECT a, NTILE(2) OVER (ORDER BY a), PERCENT_RANK() OVER (ORDER BY a) FROM u1;
SELECT a, FIRST_VALUE(c) OVER (ORDER BY a), NTH_VALUE(c, 2) OVER (ORDER BY a) FROM u1;
SELECT a, SUM(COUNT(*)) OVER () FROM u1 GROUP BY a;
SELECT (SELECT MAX(b) FROM u1) AS mx, (SELECT c FROM u1 WHERE a = 1) AS one;
SELECT a FROM u1 WHERE a IN (SELECT b FROM u1);
SELECT a FROM u1 WHERE a NOT IN (SELECT b FROM u1 WHERE b IS NOT NULL);
SELECT a FROM u1 WHERE EXISTS (SELECT 1 FROM u1 x WHERE x.a > u1.a);
SELECT a FROM u1 WHERE a > ANY (SELECT b FROM u1);
SELECT a FROM u1 WHERE a > ALL (SELECT b FROM u1 WHERE b IS NOT NULL);
SELECT * FROM (SELECT a AS k FROM u1) t WHERE k = 1;
WITH cte AS (SELECT a FROM u1), cte2 AS (SELECT a + 1 FROM cte) SELECT * FROM cte2;
WITH RECURSIVE r(n) AS (SELECT 1 UNION ALL SELECT n + 1 FROM r WHERE n < 5) SELECT SUM(n) FROM r;
SELECT a FROM u1 EXCEPT SELECT a FROM u1 WHERE a = 1;
SELECT a FROM u1 INTERSECT SELECT a FROM u1;
