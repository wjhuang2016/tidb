 c_med MEDIUMINT, c_int INT, c_intu INT UNSIGNED, c_big BIGINT, c_bigu BIGINT UNSIGNED,
 c_flt FLOAT, c_flt6 FLOAT(10,4), c_dbl DOUBLE, c_dbl6 DOUBLE(12,5),
 c_dec DECIMAL(10,4), c_bit BIT(8), c_date DATE, c_dt DATETIME(6), c_ts TIMESTAMP(3) NULL,
 c_time TIME(6), c_yr YEAR, c_ch CHAR(10), c_vc VARCHAR(100), c_bin BINARY(4),
